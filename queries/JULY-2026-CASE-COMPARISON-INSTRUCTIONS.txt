STEP 1 - COPY THE DOWNLOADED FILES INTO THE REPO

Run in PowerShell:

$repo = 'C:\AppsScript\Overview Dashboard'
$queryFolder = Join-Path $repo 'queries'
New-Item -ItemType Directory -Path $queryFolder -Force | Out-Null

$files = @(
  'DASHBOARD-CACHE-JULY-2026-CASE-EXPORT-CONSOLE.txt',
  'SSMS-JULY-2026-CASE-EXPORT.txt',
  'BIGQUERY-JULY-2026-CASE-EXPORT.txt'
)

foreach ($name in $files) {
  $source = Get-ChildItem `
    -Path "$env:USERPROFILE\Downloads" `
    -File `
    -Filter $name |
    Sort-Object LastWriteTime -Descending |
    Select-Object -First 1

  if (-not $source) {
    throw "File not found in Downloads: $name"
  }

  Copy-Item `
    -LiteralPath $source.FullName `
    -Destination (Join-Path $queryFolder $name) `
    -Force
}

Get-ChildItem -LiteralPath $queryFolder |
  Where-Object Name -in $files |
  Select-Object Name, Length, LastWriteTime

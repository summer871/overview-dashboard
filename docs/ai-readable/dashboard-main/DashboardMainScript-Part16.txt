      }, 2500);
      return true;
    } catch (primaryError) {
      try {
        link.href = 'data:' + type + ',' + encodeURIComponent(content);
        link.click();
        return true;
      } catch (fallbackError) {
        console.error('Remake CSV download failed', primaryError, fallbackError);

    } finally {

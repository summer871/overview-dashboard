    function sortMoverRows(rows, sortMode, metric) {
      const unitsMode = metric === 'units';
      const gapKey = unitsMode ? 'unitGap' : 'gap';
      const currentKey = unitsMode ? 'currentUnits' : 'currentRevenue';
      if (sortMode === 'losses') rows.sort((a,b) => a[gapKey] - b[gapKey]);
      else if (sortMode === 'gains') rows.sort((a,b) => b[gapKey] - a[gapKey]);
      else rows.sort((a,b) => b[currentKey] - a[currentKey]);
    }

    function sortRowsByColumn(rows, sortSpec) {
      const column = sortSpec && sortSpec.column;
      const dir = sortSpec && sortSpec.dir === 'desc' ? -1 : 1;
      if (!column) return;

      rows.sort((a, b) => compareSortValues(a[column], b[column]) * dir);
    }

    function compareSortValues(a, b) {
      const aNum = Number(a);
      const bNum = Number(b);
      const aIsNumber = a !== null && a !== undefined && a !== '' && !Number.isNaN(aNum);
      const bIsNumber = b !== null && b !== undefined && b !== '' && !Number.isNaN(bNum);

      if (aIsNumber && bIsNumber) {
        return aNum === bNum ? 0 : aNum > bNum ? 1 : -1;
      }

      return String(a || '').localeCompare(String(b || ''), undefined, {
        numeric: true,
        sensitivity: 'base'
      });
    }

    function sum(rows, key) {

    function groupBy(rows, keyFn, createFn, reduceFn) {
      const map = new Map();

      rows.forEach(row => {
        const key = keyFn(row);
        if (!map.has(key)) map.set(key, createFn(row));
        reduceFn(map.get(key), row);
      });

      return [...map.values()];
    }

    function getLastNonZeroIndex(values) {
      for (let i = values.length - 1; i >= 0; i--) {

    function formatChartValueLabel(value, metricKey) {
      const n = Number(value || 0);
      if (!n || !isFinite(n)) return '';

      if (metricKey === 'revenue' || metricKey === 'rpac') {
        return formatCompact(n);
      }

      if (metricKey === 'avgPrice') {
        return '$' + Math.round(n).toLocaleString('en-US');
      }

      if (metricKey === 'units' || metricKey === 'customers') {
        if (Math.abs(n) >= 1000) {
          return (n / 1000).toFixed(Math.abs(n) >= 10000 ? 0 : 1) + 'K';
        }

        return Math.round(n).toLocaleString('en-US');
      }

      return Math.round(n).toLocaleString('en-US');
    }

    function formatCurrency(value) {
      return Number(value || 0).toLocaleString('en-US', {
        style: 'currency',
        currency: 'USD',
        maximumFractionDigits: 0
      });
    }

    function formatCompact(value) {
      const n = Number(value || 0);
      if (Math.abs(n) >= 1000000) return '$' + (n / 1000000).toFixed(Math.abs(n) >= 10000000 ? 0 : 1) + 'M';
      if (Math.abs(n) >= 1000) return '$' + (n / 1000).toFixed(0) + 'K';
      return '$' + n.toFixed(0);
    }


    function formatAxisNumber(value) {
      const n = Number(value || 0);
      if (!isFinite(n)) return '0';

      const abs = Math.abs(n);
      const sign = n < 0 ? '-' : '';
      const compact = number => {
        if (number >= 1000000) {
          return (number / 1000000)
            .toFixed(number >= 10000000 ? 0 : 1)
            .replace(/\.0$/, '') + 'M';
        }
        if (number >= 1000) {
          return (number / 1000)
            .toFixed(number >= 10000 ? 0 : 1)
            .replace(/\.0$/, '') + 'K';
        }
        return Math.round(number).toLocaleString('en-US');
      };

      return sign + compact(abs);
    }

    function formatNumber(value) {
      return Number(value || 0).toLocaleString('en-US', { maximumFractionDigits: 0 });
    }

    function formatSignedPercent(value) {
      if (value === null || value === undefined || !isFinite(Number(value))) return '-';
      const rounded = Math.round(Number(value) * 100);
      return (rounded > 0 ? '+' : '') + rounded + '%';
    }

    function formatPercent(value) {
      if (value === null || value === undefined || !isFinite(Number(value))) return '-';
      return Number(value).toLocaleString('en-US', {
        style: 'percent',
        maximumFractionDigits: 1
      });
    }

    function escapeHtml(value) {
      return String(value === null || value === undefined ? '' : value)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }

    function escapeJs(value) {
      return String(value === null || value === undefined ? '' : value)
        .replace(/\\/g, '\\\\')
        .replace(/'/g, "\\'")
        .replace(/\n/g, '\\n')
        .replace(/\r/g, '\\r');
    }

    function sortByCheckedThenLabel(items, isChecked, labelFn) {
      return items.slice().sort((a, b) => {
        const checkedDiff = (isChecked(b) ? 1 : 0) - (isChecked(a) ? 1 : 0);
        if (checkedDiff) return checkedDiff;
        return String(labelFn(a) || '').localeCompare(String(labelFn(b) || ''));
      });

    /* v6.147 clickable outside-context dropdown rows

       the current dashboard result. If a product/customer sits outside the current
       Department/Product/Customer intersection, clicking it must expand the linked
       context enough for that row to become part of the result. Otherwise the click
       looks ignored because the next render still cannot see that option. */

    function getDepartmentFromProductKeyForClick(key) {
      const option = productOptions.find(item => item.key === key);
      if (option && option.department) return option.department;
      if (typeof getProductDepartmentFromKey === 'function') return getProductDepartmentFromKey(key);
      return '';
    }

    function includeDepartmentForDropdownClick(department) {
      if (!department) return;

      if (typeof selectedDepartmentsEqual === 'function' && selectedDepartmentsEqual(getAllDepartments())) state.deptMode = 'ALL';
      else if (typeof selectedDepartmentsEqual === 'function' && selectedDepartmentsEqual(getMainDepartments())) state.deptMode = 'MAIN';
      else state.deptMode = 'CUSTOM';

    function includeProductForDropdownClick(key) {
      if (!key) return;
      if (!state.selectedProductKeys) state.selectedProductKeys = new Set();
      if (!state.excludedProductKeys) state.excludedProductKeys = new Set();

        state.selectedProductKeys.add(key);

      if (typeof restoreDepartmentChildren === 'function') restoreDepartmentChildren(getDepartmentFromProductKeyForClick(key));

    }






    /* v6.149 clickable sticky dropdown selections
       Fix: rows that were visible but outside the current result could appear to ignore clicks.
       The dropdowns still show all options, but when an unchecked Product/Customer is clicked,
       the item is added to the explicit selection model and enough related context is opened
       so the checked state can persist on the next render. */

    function addDropdownSnapshotRow(snapshot, row) {
      if (!snapshot || !row) return;
      if (row.productsDepartment !== null && row.productsDepartment !== undefined && String(row.productsDepartment).trim() !== '') {
        snapshot.departments.add(row.productsDepartment);
      }
      const pKey = productKey(row);

        snapshot.products.add(pKey);
      }
      if (row.customerId !== null && row.customerId !== undefined && String(row.customerId).trim() !== '') {
        snapshot.customers.add(row.customerId);
      }
    }

    function ensureProductCustomSelectionFromSnapshot(snapshot) {
      if (state.productFilterMode === 'CUSTOM') {
        if (!state.selectedProductKeys) state.selectedProductKeys = new Set();
        return;
      }
      state.productFilterMode = 'CUSTOM';
      state.selectedProductKeys = new Set(snapshot && snapshot.products ? snapshot.products : []);
      state.excludedProductKeys = new Set();
      state.tableSelectedProductKeys = new Set(state.selectedProductKeys);
    }

    function ensureCustomerCustomSelectionFromSnapshot(snapshot) {
      if (state.customerFilterMode === 'CUSTOM') {
        if (!state.selectedCustomerIds) state.selectedCustomerIds = new Set();
        return;
      }
      state.customerFilterMode = 'CUSTOM';
      state.selectedCustomerIds = new Set(snapshot && snapshot.customers ? snapshot.customers : []);
      state.excludedCustomerIds = new Set();
      state.tableSelectedCustomerIds = new Set(state.selectedCustomerIds);
    }

    function addRelatedContextForProductSelection(key) {
      const rows = getRowsForProductKeyForDropdown(key);
      if (!state.selectedDepartments) state.selectedDepartments = new Set();
      if (!state.excludedCustomerIds) state.excludedCustomerIds = new Set();
      if (!state.selectedCustomerIds) state.selectedCustomerIds = new Set();

      rows.forEach(row => {
        if (row.productsDepartment) state.selectedDepartments.add(row.productsDepartment);
        if (row.customerId) {
          if (state.customerFilterMode === 'CUSTOM') state.selectedCustomerIds.add(row.customerId);
          else state.excludedCustomerIds.delete(row.customerId);
        }
      });

      if (typeof selectedDepartmentsEqual === 'function') {
        if (selectedDepartmentsEqual(getAllDepartments())) state.deptMode = 'ALL';
        else if (selectedDepartmentsEqual(getMainDepartments())) state.deptMode = 'MAIN';
        else state.deptMode = 'CUSTOM';
      } else {
        state.deptMode = 'CUSTOM';
      }
    }

    function addRelatedContextForCustomerSelection(id) {
      const rows = getRowsForCustomerIdForDropdown(id);
      if (!state.selectedDepartments) state.selectedDepartments = new Set();
      if (!state.selectedProductKeys) state.selectedProductKeys = new Set();
      if (!state.excludedProductKeys) state.excludedProductKeys = new Set();

      rows.forEach(row => {
        if (row.productsDepartment) state.selectedDepartments.add(row.productsDepartment);
        const key = productKey(row);
        if (state.productFilterMode === 'CUSTOM') state.selectedProductKeys.add(key);
        else state.excludedProductKeys.delete(key);
      });

      if (typeof selectedDepartmentsEqual === 'function') {
        if (selectedDepartmentsEqual(getAllDepartments())) state.deptMode = 'ALL';
        else if (selectedDepartmentsEqual(getMainDepartments())) state.deptMode = 'MAIN';
        else state.deptMode = 'CUSTOM';
      } else {
        state.deptMode = 'CUSTOM';
      }
    }

    function toggleProduct(key, event) {
      if (event) {
        if (typeof event.preventDefault === 'function') event.preventDefault();
        if (typeof event.stopPropagation === 'function') event.stopPropagation();
      }
      clearMoverTablePinnedSelections();
      if (!key) return;
      if (!state.selectedProductKeys) state.selectedProductKeys = new Set();
      if (!state.excludedProductKeys) state.excludedProductKeys = new Set();

      const snapshot = buildEffectiveDropdownSnapshot();
      const isCheckedInOpenDropdown = snapshot.products && snapshot.products.has(key);

      if (isCheckedInOpenDropdown) {
        if (state.productFilterMode === 'CUSTOM') {
          state.selectedProductKeys.delete(key);
        } else {
          state.excludedProductKeys.add(key);
        }
      } else {
        // Convert the current visible/effective product result into an explicit
        // selection, then add the clicked product. This prevents outside-context
        // rows like 3Shape from being immediately filtered back out after click.
        ensureProductCustomSelectionFromSnapshot(snapshot);
        state.selectedProductKeys.add(key);
        state.excludedProductKeys.delete(key);
        addRelatedContextForProductSelection(key);
      }

      state.lastSelectedProductKey = key;
      state.tableSelectedProductKeys = new Set(state.selectedProductKeys || []);
      if (typeof normalizeProductSelectionMode === 'function') normalizeProductSelectionMode();
      if (typeof syncSelects === 'function') syncSelects();
      renderAll();
    }

    function toggleCustomer(id, event) {
      if (event) {
        if (typeof event.preventDefault === 'function') event.preventDefault();
        if (typeof event.stopPropagation === 'function') event.stopPropagation();
      }
      clearMoverTablePinnedSelections();
      if (!id) return;
      if (!state.selectedCustomerIds) state.selectedCustomerIds = new Set();
      if (!state.excludedCustomerIds) state.excludedCustomerIds = new Set();

      const snapshot = buildEffectiveDropdownSnapshot();
      const isCheckedInOpenDropdown = snapshot.customers && snapshot.customers.has(id);

      if (isCheckedInOpenDropdown) {
        if (state.customerFilterMode === 'CUSTOM') {
          state.selectedCustomerIds.delete(id);
        } else {
          state.excludedCustomerIds.add(id);
        }
      } else {
        ensureCustomerCustomSelectionFromSnapshot(snapshot);
        state.selectedCustomerIds.add(id);
        state.excludedCustomerIds.delete(id);
        addRelatedContextForCustomerSelection(id);
      }

      state.lastSelectedCustomerId = id;
      state.tableSelectedCustomerIds = new Set(state.selectedCustomerIds || []);
      if (typeof normalizeCustomerSelectionMode === 'function') normalizeCustomerSelectionMode();
      if (typeof syncSelects === 'function') syncSelects();
      renderAll();
    }

    function getLatestComparisonMonth(rows) {
      const { currentYear } = getCurrentPriorYears();
      const currentMonths = rows
        .filter(row => row.year === currentYear)
        .map(row => Number(row.monthNum || 0))
        .filter(Boolean);
      if (currentMonths.length) return Math.max(...currentMonths);
      const allMonths = rows.map(row => Number(row.monthNum || 0)).filter(Boolean);
      return allMonths.length ? Math.max(...allMonths) : 12;
    }

    function getYtdComparisonRowsThroughCurrentMonth() {
      const rows = getOverviewTwoComparisonRows();
      const latestMonth = getLatestComparisonMonth(rows);
      return rows.filter(row => Number(row.monthNum || 0) <= latestMonth);
    }

    function getOverviewTwoComparisonYears(rows) {
      const availableYears = [...new Set(rows.map(row => Number(row.year)).filter(Boolean))].sort((a, b) => b - a);
      const selectedYears = state.years ? [...state.years].map(Number).filter(year => availableYears.includes(year)).sort((a, b) => b - a) : [];
      const years = [];
      selectedYears.forEach(year => { if (!years.includes(year)) years.push(year); });
      availableYears.forEach(year => { if (years.length < 3 && !years.includes(year)) years.push(year); });
      return years.slice(0, 3);
    }

    function renderOverviewTwoYearStack() {
      const holder = document.getElementById('o2YearStackRows');
      if (!holder) return;
      const rows = getYtdComparisonRowsThroughCurrentMonth();
      const latestMonth = getLatestComparisonMonth(getOverviewTwoComparisonRows());
      const years = getOverviewTwoComparisonYears(rows);
      const totals = new Map();
      years.forEach(year => totals.set(year, 0));
      rows.forEach(row => {
        const year = Number(row.year);
        if (totals.has(year)) totals.set(year, (totals.get(year) || 0) + Number(row.netRevenue || 0));
      });

      setText('o2YearStackAsOf', latestMonth ? 'Through ' + monthLabels[latestMonth - 1] : 'Same YTD window');
      if (!years.length) {
        holder.innerHTML = '<div class="o2YearRowV150"><span class="year">-</span><span class="amount">No data</span><span class="delta">-</span></div>';
        return;
      }

      holder.innerHTML = years.map((year, idx) => {
        const amount = totals.get(year) || 0;
        const olderYear = years[idx + 1];
        const olderAmount = olderYear ? (totals.get(olderYear) || 0) : null;
        const gap = olderAmount !== null ? amount - olderAmount : null;
        const pct = olderAmount ? gap / Math.abs(olderAmount) : null;
        const deltaClass = gap === null ? '' : gap > 0 ? 'good' : gap < 0 ? 'bad' : '';
        const deltaText = gap === null ? '-' : formatSignedPercent(pct);
        return '<div class="o2YearRowV150">'
          + '<span class="year">' + year + '</span>'
          + '<span class="amount">' + formatCurrency(amount) + '</span>'
          + '<span class="delta ' + deltaClass + '">' + deltaText + '</span>'
          + '</div>';
      }).join('');
    }

    function getMonthlyYoyRows() {
      const rows = getOverviewTwoComparisonRows();
      const { currentYear, priorYear } = getCurrentPriorYears();
      const latestMonth = getLatestComparisonMonth(rows);
      const months = [];
      for (let month = 1; month <= latestMonth; month++) months.push(month);
      return months.map(month => {
        const current = rows.filter(row => row.year === currentYear && row.monthNum === month).reduce((sum, row) => sum + Number(row.netRevenue || 0), 0);
        const prior = rows.filter(row => row.year === priorYear && row.monthNum === month).reduce((sum, row) => sum + Number(row.netRevenue || 0), 0);
        const gap = current - prior;
        const pct = prior ? gap / Math.abs(prior) : null;
        return { month, current, prior, gap, pct };
      });
    }

    function renderOverviewTwoMonthlyYoYStrip() {
      const holder = document.getElementById('o2MonthlyYoyTiles');
      if (!holder) return;
      const rawRows = getMonthlyYoyRows();
      let currentRun = 0;
      let priorRun = 0;
      const rows = rawRows.map(row => {
        currentRun += Number(row.current || 0);
        priorRun += Number(row.prior || 0);
        return Object.assign({}, row, { ytdGap: currentRun - priorRun });
      });

      const latest = rows.length ? rows[rows.length - 1] : null;
      const worst = rows.length ? rows.reduce((min, row) => row.gap < min.gap ? row : min, rows[0]) : null;
      const positiveCount = rows.filter(row => row.gap > 0).length;
      const maxAbsYtdGap = Math.max(1, ...rows.map(row => Math.abs(Number(row.ytdGap || 0))));

      setPaceValue('o2PaceYtdGap', latest ? formatCompactSigned(latest.ytdGap) : '-', latest ? latest.ytdGap : 0);
      setText('o2PaceYtdGapSub', latest ? 'Through ' + monthLabels[latest.month - 1] : 'Through current month');
      setPaceValue('o2PaceWorstMonth', worst ? monthLabels[worst.month - 1] : '-', worst ? worst.gap : 0);
      setText('o2PaceWorstMonthSub', worst ? formatCompactSigned(worst.gap) + ' monthly drag' : 'Monthly drag');
      setPaceValue('o2PacePositiveMonths', rows.length ? positiveCount + ' / ' + rows.length : '-', positiveCount);

      holder.style.setProperty('--pace-month-count', String(Math.max(rows.length, 6)));
      if (!rows.length) {
        holder.innerHTML = '<div class="o2PaceEmptyV156">No monthly comparison data available.</div>';
        return;
      }

      holder.innerHTML = rows.map(row => {
        const monthlyClass = row.gap > 0 ? 'good' : row.gap < 0 ? 'bad' : 'neutral';
        const ytdClass = row.ytdGap > 0 ? 'good' : row.ytdGap < 0 ? 'bad' : 'neutral';
        const pct = row.pct === null ? '-' : formatSignedPercent(row.pct);
        const ytdBarWidth = Math.max(8, Math.round(Math.abs(Number(row.ytdGap || 0)) / maxAbsYtdGap * 100));
        const title = monthLabels[row.month - 1] + ': monthly ' + formatCompactSigned(row.gap) + ', cumulative gap ' + formatCompactSigned(row.ytdGap);
        return '<div class="o2PaceMonthV156 o2PaceMonthV158 ' + monthlyClass + '" title="' + escapeHtml(title) + '">'
          + '<div class="o2PaceMonthLabelV156 o2PaceMonthLabelV158">' + monthLabels[row.month - 1] + '</div>'
          + '<div class="o2PaceMonthlyV156 o2PaceMonthlyV158 ' + monthlyClass + '"><span>' + pct + '</span></div>'
          + '<div class="o2PaceYtdV156 o2PaceYtdV158 ' + ytdClass + '"><b>' + formatCompactSigned(row.ytdGap) + '</b><div class="o2PaceBarTrackV156 o2PaceBarTrackV158"><i style="width:' + ytdBarWidth + '%"></i></div></div>'
          + '</div>';
      }).join('');
    }

    function drawOverviewTwoGapPath() {
      const canvasId = 'o2GapPath';
      const canvas = document.getElementById(canvasId);

      const { currentYear, priorYear } = getCurrentPriorYears();
      const latestMonth = getLatestComparisonMonth(rows);
      const months = [];
      for (let month = 1; month <= latestMonth; month++) months.push(month);

      let currentRun = 0;
      let priorRun = 0;
      const gaps = months.map(month => {
        currentRun += rows.filter(row => row.year === currentYear && row.monthNum === month).reduce((sum, row) => sum + Number(row.netRevenue || 0), 0);
        priorRun += rows.filter(row => row.year === priorYear && row.monthNum === month).reduce((sum, row) => sum + Number(row.netRevenue || 0), 0);
        return currentRun - priorRun;
      });

      if (charts[canvasId]) charts[canvasId].destroy();
      charts[canvasId] = new Chart(canvas.getContext('2d'), {
        type: 'line',
        data: {
          labels: months.map(month => monthLabels[month - 1]),
          datasets: [{
            label: 'YTD gap',
            data: gaps,
            borderColor: '#b42318',
            backgroundColor: 'rgba(180, 35, 24, 0.08)',
            pointRadius: 3,
            pointHoverRadius: 5,
            pointBackgroundColor: gaps.map(value => value >= 0 ? '#067647' : '#b42318'),
            pointBorderColor: '#ffffff',
            pointBorderWidth: 1.5,
            tension: 0.32,
            fill: true
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          interaction: { mode: 'index', intersect: false },
          onClick: (event, elements) => {
            if (!elements || !elements.length) return;
            const index = elements[0].index;
            const month = months[index];
            if (month) applyMonthSelection(month, isAdditiveEvent(event), isRangeEvent(event));
          },
          plugins: {

    function renderOverviewTwoCustomerHealth(rows) {
      const holder = document.getElementById('o2CustomerHealth');
      if (!holder) return;
      const grouped = groupCurrentPrior(rows, row => row.customerId || row.customerName || 'Unassigned', row => row.customerName || row.customerId || 'Unassigned customer');
      const growing = grouped.filter(item => item.current > item.prior && item.prior > 0);
      const declining = grouped.filter(item => item.current < item.prior && item.current > 0 && item.prior > 0);
      const newOrReactive = grouped.filter(item => item.current > 0 && item.prior <= 0);
      const lost = grouped.filter(item => item.current <= 0 && item.prior > 0);
      const decliningGap = declining.reduce((total, item) => total + Math.min(0, item.current - item.prior), 0);
      const lostPrior = lost.reduce((total, item) => total + item.prior, 0);
      const newRevenue = newOrReactive.reduce((total, item) => total + item.current, 0);
      const growingGap = growing.reduce((total, item) => total + Math.max(0, item.current - item.prior), 0);
      holder.innerHTML = [
        healthTileHtml('Growing', growing.length, formatCompactSigned(growingGap), 'good'),
        healthTileHtml('Declining', declining.length, formatCompactSigned(decliningGap), 'bad'),
        healthTileHtml('New / reactivated', newOrReactive.length, formatCurrency(newRevenue), 'good'),
        healthTileHtml('Lost', lost.length, formatCurrency(lostPrior) + ' prior', 'bad')
      ].join('');
    }

    function renderOverviewTwoBiggestMoves(rows) {
      const holder = document.getElementById('o2BiggestMoves');
      if (!holder) return;
      const customers = groupCurrentPrior(rows, row => row.customerId || row.customerName || 'Unassigned', row => row.customerName || row.customerId || 'Unassigned customer');
      const products = groupCurrentPrior(rows, row => productKey(row), row => row.productsDescription || 'Unassigned product');
      customers.forEach(item => item.gap = item.current - item.prior);
      products.forEach(item => item.gap = item.current - item.prior);
      const bestCustomer = customers.length ? customers.slice().sort((a, b) => b.gap - a.gap)[0] : null;
      const worstCustomer = customers.length ? customers.slice().sort((a, b) => a.gap - b.gap)[0] : null;
      const bestProduct = products.length ? products.slice().sort((a, b) => b.gap - a.gap)[0] : null;
      const worstProduct = products.length ? products.slice().sort((a, b) => a.gap - b.gap)[0] : null;
      const rowsOut = [
        { label: 'Best customer', item: bestCustomer },
        { label: 'Largest customer loss', item: worstCustomer },
        { label: 'Best product', item: bestProduct },
        { label: 'Largest product loss', item: worstProduct }
      ];
      holder.innerHTML = rowsOut.map(row => {
        if (!row.item) return insightRowHtml(row.label, '-', 0);

          + '</div>';
      }).join('');
    }

    function groupCurrentPrior(rows, keyFn, nameFn) {
      const { currentYear, priorYear } = getCurrentPriorYears();
      const map = new Map();
      rows.forEach(row => {
        const key = String(keyFn(row) || 'Unassigned');
        if (!map.has(key)) map.set(key, { key, name: nameFn(row) || key, current: 0, prior: 0 });
        const item = map.get(key);
        if (row.year === currentYear) item.current += Number(row.netRevenue || 0);
        if (row.year === priorYear) item.prior += Number(row.netRevenue || 0);
      });
      return [...map.values()];
    }

    function insightToneClass(value) {
      const n = Number(value || 0);
      return n > 0 ? 'good' : n < 0 ? 'bad' : '';
    }

        + '<span class="value ' + insightToneClass(toneValue) + '">' + escapeHtml(value) + '</span>'
        + '</div>';
    }

    function buildOverviewTwoInsightContext(rows, metrics) {
      const monthly = getMonthlyYoyRows();
      const customers = groupCurrentPrior(rows, row => row.customerId || row.customerName || 'Unassigned', row => row.customerName || row.customerId || 'Unassigned customer');
      const products = groupCurrentPrior(rows, row => productKey(row), row => row.productsDescription || 'Unassigned product');
      const departments = groupCurrentPrior(rows, row => row.productsDepartment || 'Unassigned', row => row.productsDepartment || 'Unassigned');
      [customers, products, departments].forEach(list => list.forEach(item => {
        item.gap = Number(item.current || 0) - Number(item.prior || 0);
        item.gapPct = item.prior ? item.gap / Math.abs(item.prior) : null;
        item.absGap = Math.abs(item.gap || 0);
      }));

      const monthWorst = monthly.length ? monthly.slice().sort((a, b) => a.gap - b.gap)[0] : null;
      const monthBest = monthly.length ? monthly.slice().sort((a, b) => b.gap - a.gap)[0] : null;
      const lossCustomers = customers.filter(item => item.gap < 0).sort((a, b) => a.gap - b.gap);
      const gainCustomers = customers.filter(item => item.gap > 0).sort((a, b) => b.gap - a.gap);
      const lossProducts = products.filter(item => item.gap < 0).sort((a, b) => a.gap - b.gap);
      const gainProducts = products.filter(item => item.gap > 0).sort((a, b) => b.gap - a.gap);
      const lossDepartments = departments.filter(item => item.gap < 0).sort((a, b) => a.gap - b.gap);
      const gainDepartments = departments.filter(item => item.gap > 0).sort((a, b) => b.gap - a.gap);
      const lostCustomers = customers.filter(item => Number(item.current || 0) <= 0 && Number(item.prior || 0) > 0);
      const newCustomers = customers.filter(item => Number(item.current || 0) > 0 && Number(item.prior || 0) <= 0);
      const growingCustomers = customers.filter(item => item.gap > 0 && item.prior > 0);
      const decliningCustomers = customers.filter(item => item.gap < 0 && item.current > 0 && item.prior > 0);
      const atRiskRevenue = lostCustomers.reduce((sum, item) => sum + Number(item.prior || 0), 0);
      const winRevenue = gainCustomers.reduce((sum, item) => sum + Math.max(0, Number(item.gap || 0)), 0);
      const lossRevenue = lossCustomers.reduce((sum, item) => sum + Math.min(0, Number(item.gap || 0)), 0);

      return {
        metrics,
        monthly,
        monthWorst,
        monthBest,
        customers,
        products,
        departments,
        lossCustomers,
        gainCustomers,
        lossProducts,
        gainProducts,
        lossDepartments,
        gainDepartments,
        lostCustomers,
        newCustomers,
        growingCustomers,
        decliningCustomers,
        atRiskRevenue,
        winRevenue,
        lossRevenue
      };
    }

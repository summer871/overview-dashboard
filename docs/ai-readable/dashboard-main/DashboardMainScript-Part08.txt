      const department = String(remakeRowValue(row, ['department']) || 'Unassigned');
      const product = String(remakeRowValue(row, ['productName', 'productId']) || 'Unknown product');
      const customer = String(remakeRowValue(row, ['customerName', 'customerId']) || 'Unknown customer');
      const reason = String(remakeRowValue(row, ['remakeReason', 'reason']) || 'Not specified');
      if (f.year && year !== f.year) return false;
      if (f.department && department !== f.department) return false;
      if (f.product && product !== f.product) return false;
      if (f.customer && customer !== f.customer) return false;
      if (f.reason && reason !== f.reason) return false;
      return true;
    });
  }

  function summarizeRemakeRows(rows) {
    const totalCases = new Set();
    const remakeCases = new Set();
    let totalUnits = 0;
    let remakeUnits = 0;
    let remakeDiscount = 0;

    rows.forEach(row => {
      const caseId = String(remakeRowValue(row, ['caseId', 'caseID', 'caseNumber']) || '').trim();
      if (caseId) totalCases.add(caseId);
      const units = Number(row.quantity || row.units || 0) || 0;
      const isRemake = !!row.isRemake || String(row.remakeFlag || row.remake || '').toLowerCase().match(/^(y|yes|true|1|r|remake)$/);
      totalUnits += units;
      if (isRemake) {
        if (caseId) remakeCases.add(caseId);
        remakeUnits += Number(row.remakeUnits || units || 0) || 0;
        remakeDiscount += Number(row.remakeDiscount || 0) || 0;
      }
    });

    return {
      totalCases: totalCases.size,
      remakeCases: remakeCases.size,
      totalUnits,
      remakeUnits,
      remakeDiscount,
      remakeCaseRate: totalCases.size ? remakeCases.size / totalCases.size : 0,
      remakeUnitRate: totalUnits ? remakeUnits / totalUnits : 0
    };
  }

  function groupRemakeRows(rows, keyGetter) {
    const groups = new Map();
    rows.forEach(row => {
      const key = String(keyGetter(row) || 'Not specified');
      if (!groups.has(key)) groups.set(key, []);
      groups.get(key).push(row);
    });
    return [...groups.entries()].map(([key, groupRows]) => ({ key, ...summarizeRemakeRows(groupRows) }));
  }

  function buildCacheV6197() {
    const years = selectedYearSetV6197();
    const factCount = Array.isArray(facts) ? facts.length : 0;
    const key = factCount + ':' + Array.from(years).sort((a, b) => a - b).join('|');

    if (cacheV6197 && cacheKeyV6197 === key) return cacheV6197;

    const departmentMap = new Map();
    const productSet = new Set();
    const customerSet = new Set();
    const yearMap = new Map();

    (facts || []).forEach(row => {
      const year = Number(row.year);
      if (!Number.isFinite(year)) return;

      if (!yearMap.has(year)) yearMap.set(year, { year, revenue: 0 });
      yearMap.get(year).revenue += Number(row.netRevenue || 0);

      if (!years.has(year)) return;

      const dept = row.productsDepartment || '(Blank department)';
      if (!departmentMap.has(dept)) {
        departmentMap.set(dept, {
          department: dept,
          isMainDepartment: !!row.isMainDepartment,
          revenue: 0
        });
      }
      departmentMap.get(dept).revenue += Number(row.netRevenue || 0);

      const pKey = productKey(row);
      if (pKey) productSet.add(pKey);

      if (row.customerId) customerSet.add(row.customerId);
    });

    cacheKeyV6197 = key;
    cacheV6197 = {
      years,
      yearsRows: Array.from(yearMap.values()).sort((a, b) => b.year - a.year),
      departments: Array.from(departmentMap.values()).sort((a, b) => Number(b.revenue || 0) - Number(a.revenue || 0)),
      productSet,
      customerSet

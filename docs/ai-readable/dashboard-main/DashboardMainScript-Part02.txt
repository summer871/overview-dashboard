    let productOptions = [];

    function buildProductControls() {
      const products = new Map();

      facts.forEach(row => {
        if (!products.has(productKey(row))) {
          products.set(productKey(row), row);
        }
      });

      productOptions = [...products.entries()]
        .map(([key, row]) => ({
          key,
          label: row.productsDescription + ' — ' + row.productsDepartment,
          department: row.productsDepartment,
          productId: row.productId,
          searchText: (row.productsDescription + ' ' + row.productsDepartment + ' ' + row.productId).toLowerCase()
        }))
        .sort((a,b) => a.label.localeCompare(b.label));

      productLabelByKey = new Map(productOptions.map(item => [item.key, item.label + ' [' + item.productId + ']']));
    }


    function getDisplayedProductOptions() {
      const affinityProducts = getProductAffinitySet();

      if (!state.collapseSelectedProducts) return rows;
      return rows.filter(item => !isProductCheckedForDropdown(item.key, affinityProducts));
    }


    function buildCustomerControls() {
      const customers = new Map();

      facts.forEach(row => {
        if (!customers.has(row.customerId)) {
          customers.set(row.customerId, {
            name: row.customerName,
            active: row.customerActive !== false
          });
        } else if (row.customerActive === false) {
          customers.get(row.customerId).active = false;
        }
      });

      customerOptions = [...customers.entries()]
        .map(([idValue, info]) => {
          const name = info && info.name ? info.name : idValue;
          const inactiveText = info && info.active === false ? ' inactive' : '';
          return {
            idValue,
            label: String(name || idValue),
            active: !(info && info.active === false),
            searchText: (String(name || '') + ' ' + String(idValue || '') + inactiveText).toLowerCase()
          };
        })
        .sort((a,b) => a.label.localeCompare(b.label));

      customerLabelById = new Map(customerOptions.map(item => [item.idValue, item.label]));
    }

    function normalizeEmptyCustomFilters() {
      // Keep Set-backed state safe without forcing sibling dropdowns to empty.
      // A cleared dashboard must be recoverable by checking one department,
      // product, or customer.
      if (!state.selectedDepartments) state.selectedDepartments = new Set();
      if (!state.selectedProductKeys) state.selectedProductKeys = new Set();
      if (!state.excludedProductKeys) state.excludedProductKeys = new Set();

    function getFilteredFacts(ignore = {}) {
      return facts.filter(row => {
        if (!ignore.years && !state.years.has(row.year)) return false;
        if (!ignore.departments && !state.selectedDepartments.has(row.productsDepartment)) return false;
        if (!ignore.product && state.productFilterMode === 'CUSTOM' && !state.selectedProductKeys.has(productKey(row))) return false;
        if (!ignore.product && state.productFilterMode !== 'CUSTOM' && state.excludedProductKeys && state.excludedProductKeys.has(productKey(row))) return false;
        if (!ignore.customer && state.customerActiveOnly && row.customerActive === false) return false;
        if (!ignore.customer && state.customerFilterMode === 'CUSTOM' && !state.selectedCustomerIds.has(row.customerId)) return false;
        if (!ignore.customer && state.customerFilterMode !== 'CUSTOM' && state.excludedCustomerIds && state.excludedCustomerIds.has(row.customerId)) return false;
        if (!ignore.month && hasMonthFilter() && !state.selectedMonths.has(row.monthNum)) return false;
        return true;
      });
    }

    function summarize(rows) {
      const { currentYear, priorYear } = getCurrentPriorYears();
      const currentRows = rows.filter(row => row.year === currentYear);
      const priorRows = rows.filter(row => row.year === priorYear);

      const revenue = sum(currentRows, 'netRevenue');
      const priorRevenue = sum(priorRows, 'netRevenue');
      const units = sum(currentRows, 'units');
      const priorUnits = sum(priorRows, 'units');
      const customers = distinctCount(currentRows, 'customerId');
      const priorCustomers = distinctCount(priorRows, 'customerId');
      const avgPrice = units ? revenue / units : 0;
      const priorAvgPrice = priorUnits ? priorRevenue / priorUnits : 0;
      const rpac = customers ? revenue / customers : 0;
      const priorRpac = priorCustomers ? priorRevenue / priorCustomers : 0;
      const gap = revenue - priorRevenue;

      return {
        revenue,
        priorRevenue,
        gap,
        gapPct: priorRevenue ? gap / Math.abs(priorRevenue) : null,
        units,
        priorUnits,
        customers,
        priorCustomers,
        rpac,
        priorRpac,
        avgPrice,
        priorAvgPrice
      };
    }

    function shouldTreatEmptyEffectiveSetAsNone(kind, effectiveSet) {
      // When a linked filter has been cleared, getAvailable* can be empty while
      // the full dropdown inventory is still displayed. In that zero-result

      if (!effectiveSet || effectiveSet.size) return false;

        return isEmptyCustomProductFilter() || isEmptyCustomCustomerFilter();

      if (kind === 'products') {
        return isEmptyDepartmentFilter() || isEmptyCustomCustomerFilter();
      }
      if (kind === 'customers') {

      }
      return false;
    }

    function hasOptionAffinityContext(ignore = {}) {
      const hasProductFilter = !ignore.product && ((state.productFilterMode === 'CUSTOM' && getSelectedProductKeys().length > 0) || (state.productFilterMode !== 'CUSTOM' && getExcludedProductKeys().length > 0));
      const hasCustomerFilter = !ignore.customer && ((state.customerFilterMode === 'CUSTOM' && getSelectedCustomerIds().length > 0) || (state.customerFilterMode !== 'CUSTOM' && getExcludedCustomerIds().length > 0));
      const mainDeptSet = new Set(getMainDepartments());
      const selectedDeptSet = state.selectedDepartments || new Set();
      const hasCustomDepartmentFilter = selectedDeptSet.size > 0 && selectedDeptSet.size < getAllDepartments().length;
      const hasMainDepartmentContext = state.deptMode === 'MAIN' && mainDeptSet.size > 0 && setsEqual(selectedDeptSet, mainDeptSet);
      const hasDepartmentFilter = !ignore.departments && (hasCustomDepartmentFilter || hasMainDepartmentContext);
      const hasYearFilter = !ignore.years && state.years && state.years.size > 0 && !isDefaultYearSelection();
      const hasMonthContext = !ignore.month && hasMonthFilter();
      return !!(hasProductFilter || hasCustomerFilter || hasDepartmentFilter || hasYearFilter || hasMonthContext);
    }

    function buildOptionAffinitySet(ignore, keyFn) {
      if (!hasOptionAffinityContext(ignore)) return new Set();
      const result = new Set();
      getFilteredFacts(ignore).forEach(row => {
        const key = keyFn(row);
        if (key !== null && key !== undefined && key !== '') result.add(key);
      });
      return result;
    }

    function getDepartmentAffinitySet() {
      return buildOptionAffinitySet({ departments: true }, row => row.productsDepartment);
    }

    function hasDepartmentAffinityContext() {
      const selectedDeptSet = state.selectedDepartments || new Set();
      if (!selectedDeptSet.size) return false;
      if (state.deptMode === 'MAIN') return true;
      if (state.deptMode === 'CUSTOM') return true;
      const allDeptSet = new Set(getAllDepartments());
      return !setsEqual(selectedDeptSet, allDeptSet);
    }

    function rowMatchesDepartmentContext(row) {
      if (!hasDepartmentAffinityContext()) return false;
      return state.selectedDepartments && state.selectedDepartments.has(row.productsDepartment);
    }

    function rowMatchesYearMonthContext(row) {
      if (state.years && state.years.size && !state.years.has(row.year)) return false;
      if (hasMonthFilter() && !state.selectedMonths.has(row.monthNum)) return false;
      return true;
    }

    function getProductAffinitySet() {
      const result = new Set();

      // Department buttons are an active filter even when Products is still set to "All products".
      // Make those department-matching products visibly rise to the top of the Product dropdown.
      if (hasDepartmentAffinityContext()) {
        productOptions.forEach(item => {
          if (state.selectedDepartments && state.selectedDepartments.has(item.department)) result.add(item.key);
        });
      }

      // Also raise products touched by the current customer/year/month context.
      if (hasOptionAffinityContext({ product: true }) || result.size) {
        getFilteredFacts({ product: true }).forEach(row => {
          if (rowMatchesYearMonthContext(row)) result.add(productKey(row));
        });
      }

      return result;
    }

    function getCustomerAffinitySet() {
      const result = new Set();

      // Department buttons are an active filter even when Customers is still set to "All customers".
      if (hasDepartmentAffinityContext()) {
        facts.forEach(row => {
          if (rowMatchesDepartmentContext(row) && rowMatchesYearMonthContext(row)) result.add(row.customerId);
        });
      }

      // Also raise customers touched by current product/year/month context.
      if (hasOptionAffinityContext({ customer: true }) || result.size) {
        getFilteredFacts({ customer: true }).forEach(row => {
          if (rowMatchesYearMonthContext(row)) result.add(row.customerId);
        });
      }

      return result;
    }

    function getYearAffinitySet() {
      return buildOptionAffinitySet({ years: true }, row => row.year);
    }

    function compareOptionPriority(aKey, bKey, selectedSet, affinitySet) {
      const selectedDiff = (setHas(selectedSet, bKey) ? 1 : 0) - (setHas(selectedSet, aKey) ? 1 : 0);
      if (selectedDiff) return selectedDiff;

      const affinityDiff = (setHas(affinitySet, bKey) ? 1 : 0) - (setHas(affinitySet, aKey) ? 1 : 0);
      if (affinityDiff) return affinityDiff;

      return 0;
    }

    function renderYearList() {
      const affinityYears = getYearAffinitySet();
      const rows = [2026, 2025, 2024].map(year => {
        // Keep the year control independent from product/customer/department filters so
        // managers can always see and re-select every year option.
        const yearRows = facts.filter(row => row.year === year);
        return { year, revenue: sum(yearRows, 'netRevenue') };
      }).sort((a, b) => {
        const priorityDiff = compareOptionPriority(a.year, b.year, state.years, affinityYears);
        if (priorityDiff) return priorityDiff;
        return b.year - a.year;
      });

      const checkedRows = rows.filter(row => state.years.has(row.year));
      const uncheckedRows = rows.filter(row => !state.years.has(row.year));
      const shownCount = state.collapseSelectedYears ? uncheckedRows.length : rows.length;
      const countEl = document.getElementById('yearVisibleCount');
      if (countEl) countEl.textContent = dropdownCountText(checkedRows.length, rows.length, shownCount);
      setCollapseButton('yearCollapseBtn', state.collapseSelectedYears);

      const rowHtml = row => `
        <div class="yearRow ${state.years.has(row.year) ? 'active' : ''}" onclick="toggleYear(${row.year}, event)">
          <div class="labelSide">
            <input type="checkbox" tabindex="-1" onclick="event.preventDefault()" ${state.years.has(row.year) ? 'checked' : ''}>
            <span>${row.year}</span>
          </div>
          <span class="rowTotal">${formatCompact(row.revenue)}</span>
        </div>`;

      const parts = [];
      parts.push(selectedCollapseHeader('years', state.collapseSelectedYears, checkedRows.length, 'Selected'));
      if (!state.collapseSelectedYears) parts.push(...checkedRows.map(rowHtml));
      parts.push(...uncheckedRows.map(rowHtml));

      document.getElementById('yearList').innerHTML = parts.join('') || '<div class="emptyFilterRow">No years match that view.</div>';
    }

    function setsEqual(a, b) {
      if (!a || !b || a.size !== b.size) return false;
      for (const value of a) {
        if (!b.has(value)) return false;
      }
      return true;
    }

    function listForTooltip(items, emptyText) {
      const cleaned = (items || []).filter(item => item !== null && item !== undefined && String(item).trim() !== '').map(item => String(item));
      if (!cleaned.length) return emptyText || '';
      const maxItems = 60;
      const shown = cleaned.slice(0, maxItems);
      const extra = cleaned.length - shown.length;
      return shown.join('\n') + (extra > 0 ? '\n+' + extra + ' more' : '');
    }

    function getCustomerNameById(customerId) {
      if (customerLabelById && customerLabelById.has(customerId)) return customerLabelById.get(customerId);
      const row = facts.find(f => f.customerId === customerId);
      return row ? row.customerName : customerId;
    }


    function renderKpis() {
      const metrics = summarize(getFilteredFacts());
      const { priorYear } = getCurrentPriorYears();

      setComparisonKpi('miniRevenue', metrics.revenue, metrics.priorRevenue, formatCurrency, priorYear);
      setComparisonKpi('miniCustomers', metrics.customers, metrics.priorCustomers, formatNumber, priorYear);
      setComparisonKpi('miniUnits', metrics.units, metrics.priorUnits, formatNumber, priorYear);
      setComparisonKpi('miniAvg', metrics.avgPrice, metrics.priorAvgPrice, formatCurrency, priorYear);
      setComparisonKpi('miniRpac', metrics.rpac, metrics.priorRpac, formatCurrency, priorYear);

      setText('bigRevenue', formatCurrency(metrics.revenue));
      setText('bigPrior', formatCurrency(metrics.priorRevenue));
      setText('bigGap', formatCurrency(metrics.gap));
      setText('bigCustomers', formatNumber(metrics.customers));
      setText('bigAvg', formatCurrency(metrics.avgPrice));
    }

    function setComparisonKpi(baseId, currentValue, priorValue, formatter, priorYear) {
      const current = Number(currentValue || 0);
      const prior = Number(priorValue || 0);
      const gap = current - prior;
      const pct = prior ? gap / Math.abs(prior) : null;
      const deltaText = formatKpiDelta(gap, pct, formatter);

      setText(baseId, formatter(current));
      setText(baseId + 'Prior', priorYear + ': ' + formatter(prior));
      setText(baseId + 'Delta', deltaText);
      setTone(baseId + 'Delta', gap);
    }

    function formatKpiDelta(gap, pct, formatter) {
      const n = Number(gap || 0);
      const arrow = n > 0 ? '▲' : n < 0 ? '▼' : '•';
      const pctText = pct === null || pct === undefined || !isFinite(Number(pct)) ? '' : ' (' + formatSignedPercent(pct) + ')';
      return arrow + ' ' + formatter(n) + pctText;
    }

    function setTone(id, value) {
      const el = document.getElementById(id);
      if (!el) return;
      el.classList.remove('good', 'bad');
      el.classList.add(Number(value) >= 0 ? 'good' : 'bad');
    }

    function renderCharts() {

      const rows = getFilteredFacts({month:true});
      const months = getChartMonths(rows);
      const years = getDisplayYears();

      destroyChartsForTab('two');
      drawRevenueTrendChart('chartRevenue', rows, years);
      drawMetricChart('chartCustomers', 'bar', rows, months, years, 'customers', formatNumber);
      drawMetricChart('chartUnits', 'bar', rows, months, years, 'units', formatNumber);
      drawMetricChart('chartRpac', 'bar', rows, months, years, 'rpac', formatCurrency);
      drawMetricChart('chartAvg', 'bar', rows, months, years, 'avgPrice', formatCurrency);

      updateChartFocusStyles();
    }

    function getChartMonths(rows) {
      const selectedMonths = getSelectedMonths();
      if (selectedMonths.length && state.monthSelectionMode !== 'multi') return selectedMonths;

      const months = [...new Set(rows.map(row => row.monthNum))]
        .filter(Boolean)
        .sort((a,b) => a-b);

      return months.length ? months : [1,2,3,4,5,6];
    }

    function hasDailyRevenueRows(rows) {
      const selectedMonth = getSingleSelectedMonth();
      return !!selectedMonth && rows.some(row => row.invoiceDate && row.dayOfMonth && row.monthNum === selectedMonth);
    }

    function shouldUseDailyRevenueTrend(baseRows) {
      const selectedMonth = getSingleSelectedMonth();
      return !!selectedMonth && state.monthSelectionMode !== 'multi' && hasDailyRevenueRows(baseRows);
    }

    function getRevenueTrendRows(baseRows) {
      const selectedMonth = getSingleSelectedMonth();
      if (!shouldUseDailyRevenueTrend(baseRows)) {
        return {
          mode: 'monthly',
          rows: baseRows,
          labels: getChartMonths(baseRows).map(month => monthLabels[month - 1]),
          points: getChartMonths(baseRows)
        };
      }

      const monthRows = getFilteredFacts().filter(row => row.monthNum === selectedMonth);
      const maxDayFromData = Math.max(
        0,
        ...monthRows.map(row => Number(row.dayOfMonth || 0)).filter(Boolean)
      );
      const selectedYears = getDisplayYears();
      const fallbackMaxDay = selectedYears.length
        ? Math.min(...selectedYears.map(year => daysInMonth(year, selectedMonth)))
        : 31;
      const maxDay = Math.max(1, maxDayFromData || fallbackMaxDay);
      const points = Array.from({length: maxDay}, (_, index) => index + 1);

      return {
        mode: 'dailyCumulative',
        rows: monthRows,
        labels: points.map(day => String(day)),
        points
      };
    }

    function dailyCumulativeMetricFor(rows, year, day) {
      const scoped = rows.filter(row =>
        row.year === Number(year) &&
        row.monthNum === Number(getSingleSelectedMonth()) &&
        Number(row.dayOfMonth || 0) <= Number(day)
      );

      return {
        revenue: sum(scoped, 'netRevenue'),
        units: sum(scoped, 'units'),
        customers: distinctCount(scoped, 'customerId')
      };
    }

    function drawRevenueTrendChart(canvasId, baseRows, years) {
      const currentYear = Number(payload && payload.config ? payload.config.currentYear : 2026);
      const priorYear = Number(payload && payload.config ? payload.config.priorYear : 2025);
      const trend = getRevenueTrendRows(baseRows);
      const isDaily = trend.mode === 'dailyCumulative';
      const selectedMonths = getSelectedMonths();
      const highlightMonthlySelection = !isDaily && state.monthSelectionMode === 'multi' && selectedMonths.length > 0;

      const datasets = years.map(year => {
        const color = colors[year] || colors[priorYear] || colors[2025];
        const isCurrent = Number(year) === currentYear;
        const isPrior = Number(year) === priorYear;

        return {
          label: String(year),
          data: trend.points.map(point => {
            if (isDaily) {
              const value = dailyCumulativeMetricFor(trend.rows, year, point).revenue;
              // v6.91: for filtered customer/product views, no purchase should read as a true zero, not a missing point.
              // This keeps the line flat on the baseline instead of creating odd diagonal fills/gaps.
              return Number.isFinite(Number(value)) ? Number(value) : 0;
            }

            const value = metricFor(trend.rows, year, point).revenue;
            // v6.91: missing rows inside the selected reporting period mean zero sales for this filter selection.
            // Future/nonexistent periods are handled by getRevenueTrendRows/getChartMonths, so keep the plotted series continuous.
            return Number.isFinite(Number(value)) ? Number(value) : 0;
          }),
          borderColor: color.border,
          backgroundColor: isCurrent ? color.area : 'transparent',
          pointBackgroundColor: color.border,
          pointBorderColor: '#ffffff',
          pointBorderWidth: isCurrent ? 2 : 0,
          pointRadius: highlightMonthlySelection
            ? trend.points.map(point => selectedMonths.includes(Number(point)) ? (isCurrent ? 4 : 2.5) : (isCurrent ? 1.6 : 0))
            : (isCurrent ? (isDaily ? 1.8 : 2.8) : 0),
          pointHoverRadius: highlightMonthlySelection
            ? trend.points.map(point => selectedMonths.includes(Number(point)) ? 5 : 3)
            : (isCurrent ? 5 : 0),
          borderWidth: isCurrent ? 3.4 : isPrior ? 2.2 : 1.8,
          borderDash: isCurrent ? [] : isPrior ? [7, 5] : [2, 5],
          fill: isCurrent ? 'origin' : false,
          tension: isDaily ? 0.28 : 0.38,
          spanGaps: false
        };
      });

      drawChart(canvasId, {
        type: 'line',
        data: {
          labels: trend.labels,
          datasets
        },
        options: chartOptions(formatCurrency, 'revenue', 'line', {
          currentYear,
          priorYear,
          isRevenueTrend: true,
          trendMode: trend.mode,
          selectedMonth: getSingleSelectedMonth(),
          canvasId
        })
      });
    }

    function metricFor(rows, year, month) {
      const scoped = rows.filter(row => row.year === year && row.monthNum === month);
      const revenue = sum(scoped, 'netRevenue');
      const units = sum(scoped, 'units');
      const customers = distinctCount(scoped, 'customerId');

      return {
        revenue,
        units,
        customers,
        rpac: customers ? revenue / customers : 0,
        avgPrice: units ? revenue / units : 0
      };
    }

    function drawMetricChart(canvasId, type, rows, months, years, metricKey, formatter) {
      const currentYear = Number(payload && payload.config ? payload.config.currentYear : 2026);
      const priorYear = Number(payload && payload.config ? payload.config.priorYear : 2025);
      const isRevenueTrend = type === 'line' && metricKey === 'revenue';
      const selectedMonths = getSelectedMonths();
      const highlightMonthlySelection = state.monthSelectionMode === 'multi' && selectedMonths.length > 0;

      const datasets = years.map(year => {
        const color = colors[year] || colors[priorYear] || colors[2025];
        const isCurrent = Number(year) === currentYear;
        const isPrior = Number(year) === priorYear;

        return {
          label: String(year),
          data: months.map(month => {
            const value = metricFor(rows, year, month)[metricKey];
            if (type === 'line' && !hasMetricRows(rows, year, month)) return null;
            return value;
          }),
          borderColor: type === 'line'
            ? color.border
            : (highlightMonthlySelection
              ? months.map(month => selectedMonths.includes(Number(month)) ? color.barBorder : color.area)
              : color.barBorder),
          backgroundColor: type === 'line'
            ? (isRevenueTrend && isCurrent ? color.area : 'transparent')
            : (highlightMonthlySelection
              ? months.map(month => selectedMonths.includes(Number(month)) ? color.bar : color.area)
              : color.bar),
          hoverBackgroundColor: type === 'bar' ? color.bar : undefined,
          hoverBorderColor: type === 'bar' ? color.barBorder : undefined,
          pointBackgroundColor: type === 'line' ? color.border : color.bar,
          pointBorderColor: '#ffffff',
          pointBorderWidth: type === 'line' && isCurrent ? 2 : 0,
          pointRadius: type === 'line' ? (isRevenueTrend ? (isCurrent ? 2.8 : 0) : (isCurrent ? 3 : 2.5)) : 0,
          pointHoverRadius: type === 'line' ? (isRevenueTrend ? (isCurrent ? 5 : 0) : (isCurrent ? 4 : 3.5)) : 0,
          borderWidth: type === 'line' ? (isCurrent ? 3.4 : isPrior ? 2.2 : 1.8) : 1.2,
          borderDash: type === 'line' ? (isCurrent ? [] : isPrior ? [7, 5] : [2, 5]) : [],
          fill: isRevenueTrend && isCurrent ? 'origin' : false,
          tension: type === 'line' ? (isRevenueTrend ? 0.38 : 0.25) : 0,
          spanGaps: false,
          borderRadius: 0,
          borderSkipped: false,
          maxBarThickness: type === 'bar' ? 34 : undefined,
          categoryPercentage: type === 'bar' ? 0.72 : undefined,
          barPercentage: type === 'bar' ? 0.82 : undefined
        };
      });

      const currentDataset = datasets.find(dataset => Number(dataset.label) === currentYear);
      const currentValues = currentDataset ? currentDataset.data.filter(value => Number(value || 0) && isFinite(Number(value || 0))).map(Number) : [];
      const averageValue = currentValues.length ? currentValues.reduce((total, value) => total + value, 0) / currentValues.length : 0;

      drawChart(canvasId, {
        type,
        data: {
          labels: months.map(month => monthLabels[month - 1]),
          datasets
        },
        options: chartOptions(formatter, metricKey, type, {
          currentYear,
          priorYear,
          isRevenueTrend,
          averageValue,
          canvasId
        })
      });
    }

    function hasMetricRows(rows, year, month) {
      return rows.some(row => row.year === Number(year) && row.monthNum === Number(month));
    }

    function getMetricAxisTitle(metricKey) {
      switch (metricKey) {
        case 'revenue':
          return 'Revenue';
        case 'customers':
          return 'Customers';
        case 'units':
          return 'Units';
        case 'rpac':
          return 'Revenue / customer';
        case 'avgPrice':
          return 'Avg $ / unit';
        default:
          return '';
      }
    }

    function externalRevenueTooltip(context, currentYear, priorYear) {
      const {chart, tooltip} = context;
      const tooltipEl = getOrCreateRevenueTooltip(chart);

      if (!tooltip || tooltip.opacity === 0) {
        tooltipEl.style.opacity = 0;
        return;
      }

      const point = (tooltip.dataPoints || []).find(item => Number(item.dataset.label) === currentYear) || (tooltip.dataPoints || [])[0];
      if (!point) {
        tooltipEl.style.opacity = 0;
        return;
      }

      const data = buildRevenueTooltipData(chart, point.dataIndex, currentYear, priorYear);
      const deltaClass = data.delta >= 0 ? 'deltaGood' : 'deltaBad';
      const arrow = data.delta >= 0 ? '▲' : '▼';
      const pct = data.deltaPct === null ? '' : ' (' + formatSignedPercent(data.deltaPct) + ')';

      const olderYearRow = data.olderYear
        ? `<div class="revenueTooltipRow comparison comparisonOlder"><span>${escapeHtml(String(data.olderYear))}</span><b>${escapeHtml(formatCurrency(data.older))}</b></div>`
        : '';

      tooltipEl.innerHTML = `
        <div class="revenueTooltipTitle">${escapeHtml(data.title)}</div>
        <div class="revenueTooltipRow current"><span>${escapeHtml(String(currentYear))}</span><b>${escapeHtml(formatCurrency(data.sales))}</b></div>
        <div class="revenueTooltipRow comparison"><span>${escapeHtml(String(priorYear))}</span><b>${escapeHtml(formatCurrency(data.prior))}</b></div>
        ${olderYearRow}
        <div class="revenueTooltipRow ${deltaClass}"><span>Δ vs ${escapeHtml(String(priorYear))}</span><b>${arrow} ${escapeHtml(formatCurrency(Math.abs(data.delta)))}${escapeHtml(pct)}</b></div>
      `;

      const canvasRect = chart.canvas.getBoundingClientRect();
      const parentRect = chart.canvas.parentNode.getBoundingClientRect();
      const leftBase = canvasRect.left - parentRect.left + tooltip.caretX;
      const topBase = canvasRect.top - parentRect.top + tooltip.caretY;

      tooltipEl.style.opacity = 1;
      tooltipEl.style.left = leftBase + 'px';
      tooltipEl.style.top = topBase + 'px';

      const width = tooltipEl.offsetWidth || 166;
      const parentWidth = chart.canvas.parentNode.clientWidth || 0;
      if (parentWidth && leftBase + width + 24 > parentWidth) {
        tooltipEl.style.transform = 'translate(calc(-100% - 10px), -50%)';
      } else {
        tooltipEl.style.transform = 'translate(10px, -50%)';
      }
    }

    function getOrCreateRevenueTooltip(chart) {
      let tooltipEl = chart.canvas.parentNode.querySelector('.revenueTooltip');
      if (!tooltipEl) {
        tooltipEl = document.createElement('div');
        tooltipEl.className = 'revenueTooltip';
        chart.canvas.parentNode.appendChild(tooltipEl);
      }
      return tooltipEl;
    }

    function buildRevenueTooltipData(chart, dataIndex, currentYear, priorYear) {
      const datasets = chart.data && chart.data.datasets ? chart.data.datasets : [];
      const currentDataset = datasets.find(dataset => Number(dataset.label) === currentYear);
      const priorDataset = datasets.find(dataset => Number(dataset.label) === priorYear);
      const olderDataset = datasets
        .filter(dataset => Number(dataset.label) !== currentYear && Number(dataset.label) !== priorYear)
        .sort((a, b) => Number(b.label) - Number(a.label))[0];

      const sales = Number(currentDataset && currentDataset.data ? currentDataset.data[dataIndex] || 0 : 0);
      const prior = Number(priorDataset && priorDataset.data ? priorDataset.data[dataIndex] || 0 : 0);
      const older = Number(olderDataset && olderDataset.data ? olderDataset.data[dataIndex] || 0 : 0);
      const olderYear = olderDataset ? Number(olderDataset.label) : null;
      const delta = sales - prior;
      const deltaPct = prior ? delta / Math.abs(prior) : null;
      const trendMeta = chart.options && chart.options.plugins ? chart.options.plugins.revenueTrendMeta || {} : {};
      const label = chart.data.labels[dataIndex] || '';
      const title = trendMeta.trendMode === 'dailyCumulative' && trendMeta.selectedMonth
        ? monthLabels[Number(trendMeta.selectedMonth) - 1] + ' ' + label + " '" + String(currentYear).slice(-2)
        : label + " '" + String(currentYear).slice(-2);

      return {
        title,
        sales,
        prior,
        older,
        olderYear,
        delta,
        deltaPct
      };
    }

    function updateChartFocusStyles() {
      const activeIds = new Set(visibleChartIdsByTab[state.activeTab] || []);
      const focusId = activeIds.has(state.focusChartId) ? state.focusChartId : null;

      // When a click applies a real data-point filter (month/product/customer), every chart is
      // now showing that selected slice. Do not dim peer charts in that state; otherwise the
      // selected month bars/points look washed out even though they are part of the selection.
      const hasAppliedDataPointFilter = hasMonthFilter() || getSelectedProductKeys().length > 0 || getSelectedCustomerIds().length > 0;

      Object.values(visibleChartIdsByTab).flat().forEach(canvasId => {
        const canvas = document.getElementById(canvasId);
        const card = canvas ? canvas.closest('.card') : null;
        if (!card) return;

        card.classList.add('chartFocusCard');
        const isVisibleChart = activeIds.has(canvasId);
        const isActive = !!focusId && canvasId === focusId;
        const shouldDim = !!focusId && isVisibleChart && canvasId !== focusId && !hasAppliedDataPointFilter;

        card.classList.toggle('chartFocusActive', isActive);
        card.classList.toggle('chartFocusDim', shouldDim);
      });
    }

    function openComponentMenu(event, componentId) {
      if (event) {
        event.preventDefault();
        event.stopPropagation();
      }

      const config = componentMenuConfigs[componentId];
      if (!config) return;

      const menu = document.getElementById('componentMenu');
      const isTable = config.type === 'table';
      const hasCanvas = !!config.canvasId;

      menu.className = 'componentMenu open' + (isTable ? ' tableMenu' : '');
      menu.innerHTML = buildComponentMenuHtml(componentId, config, hasCanvas, isTable);

      document.querySelectorAll('.componentMenuBtn.active').forEach(btn => btn.classList.remove('active'));
      if (event && event.currentTarget) event.currentTarget.classList.add('active');

      const rect = event && event.currentTarget ? event.currentTarget.getBoundingClientRect() : { left: 16, bottom: 16 };
      menu.style.display = 'block';
      const menuWidth = menu.offsetWidth || 230;
      const left = Math.max(8, Math.min(rect.right - menuWidth, window.innerWidth - menuWidth - 8));
      const top = Math.max(8, Math.min(rect.bottom + 6, window.innerHeight - (menu.offsetHeight || 260) - 8));
      menu.style.left = left + 'px';
      menu.style.top = top + 'px';
    }

    function exportComponentData(componentId) {
      const config = componentMenuConfigs[componentId];
      if (!config) return;

      const rows = config.type === 'chart'
        ? getChartExportRows(config.canvasId)
        : getTableExportRows(config.tableSelector);

      if (!rows.length) {
        showStatus('No visible data to export for ' + config.title + '.', 'warn');
        return;
      }

      downloadTextFile(slugify(config.title) + '.csv', rowsToCsv(rows), 'text/csv;charset=utf-8');
      closeComponentMenu();
    }

    function getChartExportRows(canvasId) {
      const chart = charts[canvasId];
      if (!chart || !chart.data) return [];
      const datasets = chart.data.datasets || [];
      const header = ['Label'].concat(datasets.map(dataset => String(dataset.label || 'Value')));
      const rows = [header];
      (chart.data.labels || []).forEach((label, index) => {
        rows.push([String(label)].concat(datasets.map(dataset => dataset.data && dataset.data[index] !== undefined && dataset.data[index] !== null ? dataset.data[index] : '')));
      });
      return rows;
    }

    function getTableExportRows(tableSelector) {
      const tbody = document.querySelector(tableSelector);
      const table = tbody ? tbody.closest('table') : null;
      if (!table) return [];
      const header = [...table.querySelectorAll('thead th')].map(th => cleanCellText(th.textContent));
      const rows = [header];
      [...table.querySelectorAll('tbody tr')].forEach(tr => {
        rows.push([...tr.querySelectorAll('td')].map(td => cleanCellText(td.textContent)));
      });
      return rows;
    }

    function cleanCellText(value) {
      return String(value || '').replace(/▲|▼/g, '').replace(/\s+/g, ' ').trim();
    }

    function rowsToCsv(rows) {
      return rows.map(row => row.map(cell => {
        const text = String(cell === null || cell === undefined ? '' : cell);
        return '"' + text.replace(/"/g, '""') + '"';
      }).join(',')).join('\r\n');
    }

    function downloadTextFile(fileName, text, mimeType) {
      const blob = new Blob([text], { type: mimeType || 'text/plain;charset=utf-8' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = fileName;
      document.body.appendChild(a);
      a.click();
      a.remove();
      setTimeout(() => URL.revokeObjectURL(url), 1000);
    }

    async function renderComponentToPngBlob(componentId) {
      const config = componentMenuConfigs[componentId];
      if (!config) return null;

      const card = getComponentCard(componentId);
      if (card && window.html2canvas) {
        const activeMenuButtons = [...card.querySelectorAll('.componentMenuBtn.active')];
        activeMenuButtons.forEach(button => button.classList.remove('active'));
        const canvas = await html2canvas(card, {
          backgroundColor: '#ffffff',
          scale: Math.min(2, window.devicePixelRatio || 1.5),
          useCORS: true,
          logging: false,
          scrollX: 0,
          scrollY: 0
        });
        activeMenuButtons.forEach(button => button.classList.add('active'));
        return new Promise(resolve => canvas.toBlob(resolve, 'image/png'));
      }

      const canvas = config.canvasId ? document.getElementById(config.canvasId) : null;
      if (canvas && canvas.toBlob) {
        return new Promise(resolve => canvas.toBlob(resolve, 'image/png'));
      }

      return null;
    }

    function blobToObjectUrl(blob) {
      return URL.createObjectURL(blob);
    }

    async function saveComponentPng(componentId) {
      const config = componentMenuConfigs[componentId];
      if (!config) return;
      closeComponentMenu();

      try {
        showStatus('Preparing ' + config.title + ' PNG...');
        const blob = await renderComponentToPngBlob(componentId);
        if (!blob) {
          showStatus('Could not create PNG for ' + config.title + '.', 'warn');
          return;
        }
        const url = blobToObjectUrl(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = slugify(config.title) + '.png';
        document.body.appendChild(a);
        a.click();
        a.remove();
        setTimeout(() => URL.revokeObjectURL(url), 1200);
        showStatus('Downloaded ' + config.title + ' PNG.', 'good');
      } catch (error) {
        showStatus('Could not create PNG for ' + config.title + '. Try Export data instead.', 'warn');
      }
    }

    async function copyComponentImage(componentId) {
      const config = componentMenuConfigs[componentId];
      if (!config) return;
      closeComponentMenu();

      if (!navigator.clipboard || !window.ClipboardItem) {
        await saveComponentPng(componentId);
        showStatus('Image copy is not available in this browser, so the PNG was downloaded instead.', 'warn');
        return;
      }

      try {
        const blob = await renderComponentToPngBlob(componentId);
        if (!blob) {
          showStatus('Could not create image for ' + config.title + '.', 'warn');
          return;
        }
        await navigator.clipboard.write([new ClipboardItem({ 'image/png': blob })]);
        showStatus('Copied ' + config.title + ' image to clipboard.', 'good');
      } catch (error) {
        await saveComponentPng(componentId);
        showStatus('Clipboard copy was blocked, so the PNG was downloaded instead.', 'warn');
      }
    }

    function copyComponentLink(componentId) {
      const config = componentMenuConfigs[componentId];
      const card = getComponentCard(componentId);
      const hash = '#component-' + componentId;
      const url = window.location.href.split('#')[0] + hash;
      history.replaceState(null, '', hash);
      if (card) card.scrollIntoView({ behavior: 'smooth', block: 'center' });

      if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(url)
          .then(() => showStatus('Copied link to ' + (config ? config.title : 'component') + '.', 'good'))
          .catch(() => showStatus('Component link: ' + url, 'warn'));
      } else {
        showStatus('Component link: ' + url, 'warn');
      }
      closeComponentMenu();
    }

    function getComponentCard(componentId) {
      const config = componentMenuConfigs[componentId];
      if (!config) return null;
      if (config.canvasId) {
        const canvas = document.getElementById(config.canvasId);
        return canvas ? canvas.closest('.card') : null;
      }
      if (config.tableSelector) {
        const tableBody = document.querySelector(config.tableSelector);
        return tableBody ? tableBody.closest('.card') : null;
      }
      return null;
    }

    function scrollToLinkedComponent() {
      const hash = String(window.location.hash || '');
      if (!hash.startsWith('#component-')) return;
      const componentId = hash.replace('#component-', '');
      const card = getComponentCard(componentId);
      if (card) {
        card.scrollIntoView({ behavior: 'smooth', block: 'center' });
        card.classList.add('chartFocusActive');
        setTimeout(() => card.classList.remove('chartFocusActive'), 1800);
      }
    }

    function slugify(value) {
      return String(value || 'component')
        .toLowerCase()
        .replace(/[^a-z0-9]+/g, '-')
        .replace(/^-+|-+$/g, '') || 'component';
    }

    function bumpSelectedRowsToTop(rows, keyFn, selectedSet) {
      if (!selectedSet || !selectedSet.size || !Array.isArray(rows)) return rows;
      rows.forEach((row, index) => { row.__selectedSortIndex = index; });
      rows.sort((a, b) => {
        const aSelected = selectedSet.has(keyFn(a)) ? 1 : 0;
        const bSelected = selectedSet.has(keyFn(b)) ? 1 : 0;
        const selectedDiff = bSelected - aSelected;
        if (selectedDiff) return selectedDiff;
        return (a.__selectedSortIndex || 0) - (b.__selectedSortIndex || 0);
      });
      rows.forEach(row => { delete row.__selectedSortIndex; });
      return rows;
    }

    function renderCustomerTable() {
      // Keep the Customer Movers inventory visible when customers are selected.
      // Selected customers are bumped to the top, while product/department/year/month filters
      // still cross-filter this table.
      const rows = groupBy(getFilteredFacts({ customer: true }), row => row.customerId, row => ({
        customerId: row.customerId,
        customerName: row.customerName,
        customerActive: true,
        currentRevenue: 0,
        priorRevenue: 0,
        currentUnits: 0,
        priorUnits: 0
      }), (acc, row) => {
        if (row.customerActive === false) acc.customerActive = false;
        if (row.year === (payload.config.currentYear || 2026)) {
          acc.currentRevenue += row.netRevenue;
          acc.currentUnits += row.units;
        }
        if (row.year === (payload.config.priorYear || 2025)) {
          acc.priorRevenue += row.netRevenue;
          acc.priorUnits += row.units;
        }
      }).map(row => ({
        ...row,
        gap: row.currentRevenue - row.priorRevenue,
        unitGap: row.currentUnits - row.priorUnits
      }));

      updateMoverMetricHeaders('customer');
      sortCustomerRows(rows);
      bumpSelectedRowsToTop(rows, row => row.customerId, state.tableSelectedCustomerIds);

      const visibleRows = rows.slice(0, 500);
      state.lastCustomerTableIds = visibleRows.map(row => row.customerId);
      const unitsMode = state.customerMetric === 'units';

      document.getElementById('customerTable').innerHTML = visibleRows.map(row => {
        const priorValue = unitsMode ? row.priorUnits : row.priorRevenue;
        const currentValue = unitsMode ? row.currentUnits : row.currentRevenue;
        const gapValue = unitsMode ? row.unitGap : row.gap;
        return `
          <tr class="clickable ${state.tableSelectedCustomerIds && state.tableSelectedCustomerIds.has(row.customerId) ? 'active' : ''}" onclick="setCustomer('${escapeJs(row.customerId)}', event)">
            <td>${escapeHtml(row.customerId)}</td>
            <td class="customerNameCell"><span class="customerNameText">${escapeHtml(row.customerName)}</span>${row.customerActive === false ? ' <span class="inactiveBadge">Inactive</span>' : ''}</td>
            <td>${formatMoverValue(priorValue, unitsMode)}</td>
            <td>${formatMoverValue(currentValue, unitsMode)}</td>
            <td class="${gapValue < 0 ? 'negative' : 'positive'}">${formatMoverValue(gapValue, unitsMode)}</td>
          </tr>
        `;
      }).join('');
    }

(function(){
  'use strict';
  const tableId = 'remakeCeramistTableV6342';
  const pinnedClass = 'ceramistPinnedWorkerV6361';
  let observedTable = null;
  let tableObserver = null;
  let resizeObserver = null;
  let queued = false;

  function table(){
    return document.getElementById(tableId);
  }

  function setPinTop(node, pixels){
    const next = pixels + 'px';
    if (node.style.getPropertyValue('--ceramist-selected-pin-top-v6361') !== next) {
      node.style.setProperty('--ceramist-selected-pin-top-v6361', next);
    }
  }

  function removePinnedRows(node, exceptRow){
    node.querySelectorAll('tbody tr.' + pinnedClass).forEach(function(row){
      if (row !== exceptRow) row.classList.remove(pinnedClass);
    });
  }

  function measureAndPin(){
    queued = false;
    const node = table();
    if (!node) return;

    const mobile = !!(window.matchMedia && window.matchMedia('(max-width:760px)').matches);
    const selected = mobile ? null : node.querySelector('tbody tr.ceramistSelectedV6343');

    removePinnedRows(node, selected);
    if (!selected) {
      const current = node.querySelector('tbody tr.' + pinnedClass);
      if (current) current.classList.remove(pinnedClass);
      return;
    }

    const header = node.querySelector('thead');
    const headerHeight = Math.max(1, Math.ceil(header ? header.getBoundingClientRect().height : 28));
    setPinTop(node, headerHeight);

    if (!selected.classList.contains(pinnedClass)) selected.classList.add(pinnedClass);
  }

  function queue(){
    if (queued) return;
    queued = true;
    if (window.requestAnimationFrame) window.requestAnimationFrame(measureAndPin);
    else window.setTimeout(measureAndPin, 0);
  }

  function observe(){
    const node = table();
    if (!node || node === observedTable) {
      queue();
      return;
    }

    if (tableObserver) tableObserver.disconnect();
    if (resizeObserver) resizeObserver.disconnect();

    observedTable = node;
    tableObserver = new MutationObserver(queue);
    tableObserver.observe(node, {

      subtree:true,
      attributes:true,

    });

    if (window.ResizeObserver) {

      const header = node.querySelector('thead');
      if (header) resizeObserver.observe(header);
    }
